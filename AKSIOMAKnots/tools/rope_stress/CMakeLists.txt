cmake_minimum_required(VERSION 3.21)
project(rope_stress LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(rope_stress rope_stress.cpp
    ${SRC}/knotmath.cpp ${SRC}/knottable.cpp ${SRC}/photoscene.cpp ${SRC}/ropephysics.cpp)
target_include_directories(rope_stress PRIVATE ${SRC})
target_compile_options(rope_stress PRIVATE -O2 -Wall -Wextra)
qt_add_resources(rope_stress "data" PREFIX "/" BIG_RESOURCES BASE ${SRC} FILES ${SRC}/data/knottable.dat)
target_link_libraries(rope_stress PRIVATE Qt6::Core Qt6::Gui)
