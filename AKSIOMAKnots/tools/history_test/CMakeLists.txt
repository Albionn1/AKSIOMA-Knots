cmake_minimum_required(VERSION 3.21)
project(history_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(history_test history_test.cpp ${SRC}/knothistory.cpp ${SRC}/knotmath.cpp ${SRC}/knottable.cpp)
target_include_directories(history_test PRIVATE ${SRC})
target_compile_options(history_test PRIVATE -O2 -Wall -Wextra)
qt_add_resources(history_test "knottable" PREFIX "/" BIG_RESOURCES BASE ${SRC} FILES ${SRC}/data/knottable.dat)
target_link_libraries(history_test PRIVATE Qt6::Core Qt6::Gui)

find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(help_test help_test.cpp ${SRC}/commandpalette.h ${SRC}/commandpalette.cpp ${SRC}/controlhelp.h ${SRC}/controlhelp.cpp)
target_include_directories(help_test PRIVATE ${SRC})
target_compile_options(help_test PRIVATE -O2 -Wall -Wextra)
target_link_libraries(help_test PRIVATE Qt6::Widgets)
