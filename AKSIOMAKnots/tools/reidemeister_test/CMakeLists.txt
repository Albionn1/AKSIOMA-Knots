cmake_minimum_required(VERSION 3.21)
project(reidemeister_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(reidemeister_test reidemeister_test.cpp
    ${SRC}/knotmath.cpp ${SRC}/knottable.cpp ${SRC}/knotdiagram.cpp ${SRC}/knotreidemeister.cpp)
target_include_directories(reidemeister_test PRIVATE ${SRC})
target_compile_options(reidemeister_test PRIVATE -O2 -Wall -Wextra)
qt_add_resources(reidemeister_test "knottable" PREFIX "/" BIG_RESOURCES BASE ${SRC} FILES ${SRC}/data/knottable.dat)
target_link_libraries(reidemeister_test PRIVATE Qt6::Core Qt6::Gui)
