cmake_minimum_required(VERSION 3.21)
project(math_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(math_audit math_audit.cpp ${SRC}/knotmath.cpp ${SRC}/knottable.cpp)
target_include_directories(math_audit PRIVATE ${SRC})
target_compile_options(math_audit PRIVATE -O2 -Wall -Wextra)
qt_add_resources(math_audit "knottable" PREFIX "/" BIG_RESOURCES BASE ${SRC} FILES ${SRC}/data/knottable.dat)
target_link_libraries(math_audit PRIVATE Qt6::Core Qt6::Gui)
